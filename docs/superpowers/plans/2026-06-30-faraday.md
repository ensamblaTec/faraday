# Faraday Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Una CLI en Go (`faraday`) que ejecuta una app con la red **totalmente bloqueada** (sin root), al estilo firejail, con perfiles por app.

**Architecture:** Una interfaz `sandbox.Runner` con 3 backends — Go-puro (`CLONE_NEWUSER|CLONE_NEWNET` + re-exec del child), `bwrap` (fallback) y `flatpak` (`--unshare=network`) — elegidos por `Select(Spec)`. La CLI fusiona flags + perfiles YAML en un `Spec` y delega en el runner.

**Tech Stack:** Go 1.25, stdlib (`os/exec`, `syscall`), `golang.org/x/sys/unix` (Exec), `github.com/vishvananda/netlink` (subir `lo`), `gopkg.in/yaml.v3` (perfiles).

## Global Constraints

- Módulo: `github.com/ensamblatec/faraday`. `go.mod`: `go 1.25` (NO bumpear; instalado 1.26.4).
- Build siempre `CGO_ENABLED=0` (todas las deps son pure-Go).
- Plataforma: Linux only (namespaces). Tests de aislamiento usan build tag `//go:build linux`.
- **Commits: solo con autorización del maintainer**. Los pasos "Commit" definen los límites y mensajes; durante la ejecución, *stagear* y pedir luz verde antes de `git commit`.
- Mapeo UID/GID **identidad** (uid→mismo uid) para que X11/Wayland/PipeWire sigan funcionando.

```go
package cli

import (
	"fmt"
	"io"
	"os"
)

const version = "0.1.0"

// Main dispatches a faraday subcommand and returns the process exit code.
func Main(args []string) int {
	if len(args) == 0 {
		usage(os.Stderr)
		return 2
	}
	switch args[0] {
	case "run":
		return cmdRun(args[1:])
	case "list":
		return cmdList(args[1:])
	case "save":
		return cmdSave(args[1:])
	case "version", "--version", "-v":
		fmt.Println("faraday", version)
		return 0
	case "help", "--help", "-h":
		usage(os.Stdout)
		return 0
	default:
		fmt.Fprintf(os.Stderr, "faraday: unknown command %q\n", args[0])
		usage(os.Stderr)
		return 2
	}
}

func usage(w io.Writer) {
	fmt.Fprint(w, `faraday — run apps inside a Faraday cage (zero network)

usage:
  faraday run [--loopback] -- <cmd> [args...]   run a command with no network
  faraday run [--loopback] <profile>            run a saved profile
  faraday list                                  list saved profiles
  faraday save <name> [--loopback] (--flatpak <appid> | -- <cmd>...)
  faraday version | help
`)
}
```

```go
// Package profile loads and saves per-app faraday sandbox profiles (YAML).
package profile

import (
	"errors"
	"fmt"
	"os"
	"path/filepath"

	"gopkg.in/yaml.v3"
)

// Profile is a saved sandbox configuration for one app.
type Profile struct {
	Description string   `yaml:"description,omitempty"`
	Flatpak     string   `yaml:"flatpak,omitempty"` // flatpak app id; non-empty → flatpak backend
	Loopback    bool     `yaml:"loopback,omitempty"`
	Cmd         []string `yaml:"cmd,omitempty"` // command+args for non-flatpak apps
}

// Dir returns the profiles directory (~/.config/faraday/profiles), honoring XDG_CONFIG_HOME.
func Dir() (string, error) {
	base := os.Getenv("XDG_CONFIG_HOME")
	if base == "" {
		home, err := os.UserHomeDir()
		if err != nil {
			return "", err
		}
		base = filepath.Join(home, ".config")
	}
	return filepath.Join(base, "faraday", "profiles"), nil
}

// Load reads a profile by name from the profiles dir.
func Load(name string) (*Profile, error) {
	dir, err := Dir()
	if err != nil {
		return nil, err
	}
	return LoadFile(filepath.Join(dir, name+".yaml"))
}

// LoadFile reads a profile from an explicit path.
func LoadFile(path string) (*Profile, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		return nil, err
	}
	var p Profile
	if err := yaml.Unmarshal(data, &p); err != nil {
		return nil, fmt.Errorf("parse %s: %w", path, err)
	}
	return &p, nil
}

// Save writes a profile as <name>.yaml in the profiles dir.
func Save(name string, p *Profile) error {
	dir, err := Dir()
	if err != nil {
		return err
	}
	if err := os.MkdirAll(dir, 0o755); err != nil {
		return err
	}
	data, err := yaml.Marshal(p)
	if err != nil {
		return err
	}
	return os.WriteFile(filepath.Join(dir, name+".yaml"), data, 0o644)
}

// List returns the names (without extension) of available profiles.
func List() ([]string, error) {
	dir, err := Dir()
	if err != nil {
		return nil, err
	}
	entries, err := os.ReadDir(dir)
	if errors.Is(err, os.ErrNotExist) {
		return nil, nil
	}
	if err != nil {
		return nil, err
	}
	var names []string
	for _, e := range entries {
		if e.IsDir() {
			continue
		}
		ext := filepath.Ext(e.Name())
		if ext == ".yaml" || ext == ".yml" {
			names = append(names, e.Name()[:len(e.Name())-len(ext)])
		}
	}
	return names, nil
}
```

```go
// Package sandbox runs a command with its network isolated, picking a backend.
package sandbox

import (
	"context"
	"errors"
)

// Spec describes a sandboxed launch.
type Spec struct {
	Cmd          []string // command + args (non-flatpak)
	Loopback     bool     // bring up the namespace-internal lo
	Flatpak      bool     // use the flatpak backend
	FlatpakAppID string   // flatpak application id
	FlatpakArgs  []string // extra args after the app id
	ForceBwrap   bool     // skip the userns backend (testing / fallback)
	Env          []string // child environment (nil → os.Environ())
}

// Runner executes a Spec with the network blocked.
type Runner interface {
	Run(ctx context.Context, spec Spec) error
	Available() bool
	Name() string
}

// ErrNoBackend is returned when no isolation backend is usable.
var ErrNoBackend = errors.New("faraday: no usable sandbox backend (unprivileged userns disabled and bwrap not found)")

// Select picks the most appropriate backend for spec.
func Select(spec Spec) (Runner, error) {
	if spec.Flatpak {
		r := &flatpakRunner{}
		if r.Available() {
			return r, nil
		}
		return nil, errors.New("faraday: flatpak requested but flatpak not installed")
	}
	if !spec.ForceBwrap {
		if r := (&nsRunner{}); r.Available() {
			return r, nil
		}
	}
	if r := (&bwrapRunner{}); r.Available() {
		return r, nil
	}
	return nil, ErrNoBackend
}
```

```go
package sandbox

import (
	"context"
	"os"
	"os/exec"
)

type flatpakRunner struct{}

func (r *flatpakRunner) Name() string { return "flatpak" }

func (r *flatpakRunner) Available() bool {
	_, err := exec.LookPath("flatpak")
	return err == nil
}

func (r *flatpakRunner) Run(ctx context.Context, spec Spec) error {
	args := []string{"run", "--unshare=network", "--die-with-parent", spec.FlatpakAppID}
	args = append(args, spec.FlatpakArgs...)
	cmd := exec.CommandContext(ctx, "flatpak", args...)
	cmd.Stdin, cmd.Stdout, cmd.Stderr = os.Stdin, os.Stdout, os.Stderr
	cmd.Env = envOrDefault(spec.Env)
	return cmd.Run()
}

// DetectFlatpak reports whether appID is an installed flatpak.
func DetectFlatpak(appID string) bool {
	if _, err := exec.LookPath("flatpak"); err != nil {
		return false
	}
	return exec.Command("flatpak", "info", appID).Run() == nil
}

func envOrDefault(env []string) []string {
	if env == nil {
		return os.Environ()
	}
	return env
}
```

```go
package sandbox

import (
	"os"
	"os/exec"
	"sync"
	"syscall"
)

var (
	usernsOnce sync.Once
	usernsOK   bool
)

// usernsAvailable reports whether unprivileged user+net namespaces can be created.
// It probes once by re-exec'ing this binary as "__probe" inside a new userns+netns;
// reaching main with FARADAY_PROBE set means success (the child just exits 0).
func usernsAvailable() bool {
	usernsOnce.Do(func() {
		self, err := os.Executable()
		if err != nil {
			self = "/proc/self/exe"
		}
		cmd := exec.Command(self, "__probe")
		cmd.Env = []string{"FARADAY_PROBE=1"}
		cmd.SysProcAttr = cloneAttr()
		usernsOK = cmd.Run() == nil
	})
	return usernsOK
}

// cloneAttr returns the SysProcAttr that creates a new user+net namespace with an
// identity uid/gid mapping (so X11/Wayland/PipeWire credentials keep working).
func cloneAttr() *syscall.SysProcAttr {
	return &syscall.SysProcAttr{
		Cloneflags: syscall.CLONE_NEWUSER | syscall.CLONE_NEWNET,
		UidMappings: []syscall.SysProcIDMap{
			{ContainerID: os.Getuid(), HostID: os.Getuid(), Size: 1},
		},
		GidMappings: []syscall.SysProcIDMap{
			{ContainerID: os.Getgid(), HostID: os.Getgid(), Size: 1},
		},
		GidMappingsEnableSetgroups: false,
	}
}
```

```go
package sandbox

import (
	"context"
	"encoding/json"
	"fmt"
	"os"
	"os/exec"
)

// childEnvVar carries the JSON-encoded Spec to the re-exec'd child.
const childEnvVar = "FARADAY_CHILD"

type nsRunner struct{}

func (r *nsRunner) Name() string    { return "netns" }
func (r *nsRunner) Available() bool { return usernsAvailable() }

func (r *nsRunner) Run(ctx context.Context, spec Spec) error {
	if len(spec.Cmd) == 0 {
		return fmt.Errorf("faraday: empty command")
	}
	payload, err := json.Marshal(spec)
	if err != nil {
		return fmt.Errorf("faraday: marshal spec: %w", err)
	}
	self, err := os.Executable()
	if err != nil {
		self = "/proc/self/exe"
	}
	cmd := exec.CommandContext(ctx, self, "__child")
	cmd.Env = append(envOrDefault(spec.Env), childEnvVar+"="+string(payload))
	cmd.Stdin, cmd.Stdout, cmd.Stderr = os.Stdin, os.Stdout, os.Stderr
	cmd.SysProcAttr = cloneAttr()
	return cmd.Run()
}
```

- [ ] **Step 4: Implement `internal/sandbox/child.go`**

```go
package sandbox

import (
	"encoding/json"
	"fmt"
	"os"
	"os/exec"
	"strings"

	"github.com/vishvananda/netlink"
	"golang.org/x/sys/unix"
)

// RunChild runs inside the freshly-created user+net namespace. It optionally
// brings up loopback, then execs the target, replacing this process.
func RunChild() error {
	payload := os.Getenv(childEnvVar)
	if payload == "" {
		return fmt.Errorf("faraday: %s not set", childEnvVar)
	}
	var spec Spec
	if err := json.Unmarshal([]byte(payload), &spec); err != nil {
		return fmt.Errorf("faraday: decode child spec: %w", err)
	}
	if len(spec.Cmd) == 0 {
		return fmt.Errorf("faraday: empty command in child")
	}
	if spec.Loopback {
		if err := bringLoopbackUp(); err != nil {
			return fmt.Errorf("faraday: bring up lo: %w", err)
		}
	}
	path, err := exec.LookPath(spec.Cmd[0])
	if err != nil {
		return fmt.Errorf("faraday: command not found: %s", spec.Cmd[0])
	}
	return unix.Exec(path, spec.Cmd, childEnviron())
}

// bringLoopbackUp sets the namespace-internal lo interface up.
func bringLoopbackUp() error {
	lo, err := netlink.LinkByName("lo")
	if err != nil {
		return err
	}
	return netlink.LinkSetUp(lo)
}

// childEnviron returns the current environment without the FARADAY_CHILD sentinel.
func childEnviron() []string {
	all := os.Environ()
	out := make([]string, 0, len(all))
	for _, kv := range all {
		if strings.HasPrefix(kv, childEnvVar+"=") {
			continue
		}
		out = append(out, kv)
	}
	return out
}
```

```go
package cli

import (
	"context"
	"errors"
	"fmt"
	"os"
	"os/exec"
	"strings"

	"github.com/ensamblatec/faraday/internal/profile"
	"github.com/ensamblatec/faraday/internal/sandbox"
)

type runPlan struct {
	cmd         []string // ad-hoc command (after --); nil when using a profile
	profileName string   // profile to load; "" when ad-hoc
	loopback    bool     // --loopback flag
}

// splitArgs separates args before "--" from the command after it.
func splitArgs(args []string) (before, cmd []string, hasSep bool) {
	for i, a := range args {
		if a == "--" {
			return args[:i], args[i+1:], true
		}
	}
	return args, nil, false
}

// parseRun parses `run` arguments into a runPlan (no side effects).
func parseRun(args []string) (runPlan, error) {
	before, cmd, hasSep := splitArgs(args)
	var rp runPlan
	var positionals []string
	for _, a := range before {
		switch {
		case a == "--loopback":
			rp.loopback = true
		case strings.HasPrefix(a, "-"):
			return rp, fmt.Errorf("unknown flag %q", a)
		default:
			positionals = append(positionals, a)
		}
	}
	switch {
	case hasSep:
		if len(cmd) == 0 {
			return rp, errors.New("no command after --")
		}
		rp.cmd = cmd
		return rp, nil
	case len(positionals) == 1:
		rp.profileName = positionals[0]
		return rp, nil
	default:
		return rp, errors.New("usage: faraday run [--loopback] (<profile> | -- <cmd> [args...])")
	}
}

// SpecFromProfile builds a sandbox.Spec from a saved profile.
func SpecFromProfile(p *profile.Profile) sandbox.Spec {
	spec := sandbox.Spec{Loopback: p.Loopback, Cmd: p.Cmd}
	if p.Flatpak != "" {
		spec.Flatpak = true
		spec.FlatpakAppID = p.Flatpak
	}
	return spec
}

func cmdRun(args []string) int {
	rp, err := parseRun(args)
	if err != nil {
		fmt.Fprintln(os.Stderr, "faraday run:", err)
		return 2
	}
	var spec sandbox.Spec
	if rp.profileName != "" {
		p, err := profile.Load(rp.profileName)
		if err != nil {
			fmt.Fprintf(os.Stderr, "faraday run: load profile %q: %v\n", rp.profileName, err)
			return 1
		}
		spec = SpecFromProfile(p)
		if rp.loopback {
			spec.Loopback = true
		}
	} else {
		spec = sandbox.Spec{Cmd: rp.cmd, Loopback: rp.loopback}
	}

	runner, err := sandbox.Select(spec)
	if err != nil {
		fmt.Fprintln(os.Stderr, "faraday run:", err)
		return 1
	}
	if err := runner.Run(context.Background(), spec); err != nil {
		var ee *exec.ExitError
		if errors.As(err, &ee) {
			return ee.ExitCode()
		}
		fmt.Fprintln(os.Stderr, "faraday run:", err)
		return 1
	}
	return 0
}

func cmdList(args []string) int {
	names, err := profile.List()
	if err != nil {
		fmt.Fprintln(os.Stderr, "faraday list:", err)
		return 1
	}
	if len(names) == 0 {
		fmt.Println("no profiles yet (add one with: faraday save <name> ...)")
		return 0
	}
	for _, n := range names {
		fmt.Println(n)
	}
	return 0
}

func cmdSave(args []string) int {
	before, cmd, hasSep := splitArgs(args)
	p := &profile.Profile{}
	var positionals []string
	for i := 0; i < len(before); i++ {
		switch before[i] {
		case "--loopback":
			p.Loopback = true
		case "--flatpak":
			if i+1 >= len(before) {
				fmt.Fprintln(os.Stderr, "faraday save: --flatpak needs an app id")
				return 2
			}
			i++
			p.Flatpak = before[i]
		default:
			positionals = append(positionals, before[i])
		}
	}
	if len(positionals) != 1 {
		fmt.Fprintln(os.Stderr, "usage: faraday save <name> [--loopback] (--flatpak <appid> | -- <cmd>...)")
		return 2
	}
	switch {
	case p.Flatpak != "":
		// ok
	case hasSep && len(cmd) > 0:
		p.Cmd = cmd
	default:
		fmt.Fprintln(os.Stderr, "faraday save: provide --flatpak <appid> or -- <cmd>...")
		return 2
	}
	if err := profile.Save(positionals[0], p); err != nil {
		fmt.Fprintln(os.Stderr, "faraday save:", err)
		return 1
	}
	fmt.Printf("saved profile %q\n", positionals[0])
	return 0
}
```
